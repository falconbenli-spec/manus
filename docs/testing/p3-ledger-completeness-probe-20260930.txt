// Read-only probe of the measured gaps against the current ledger code, on a synthetic in-memory month.
const W=process.cwd();
const { ledgerMonth, OLD_PURPOSES } = await import(W+'/tests/ledger-fixture.mjs');
const ledger = await import(W+'/app/ledger.mjs');
const f = await import(W+'/app/finance.mjs');
const { readFileSync } = await import('node:fs');
const cleanups=[];const t={after:fn=>cleanups.push(fn)};
const out=(k,v)=>console.log(`${k}: ${typeof v==='string'?v:JSON.stringify(v)}`);
const tryCode=fn=>{try{fn();return 'no error';}catch(e){return e.code??e.message;}};
const m=ledgerMonth(t,{purposes:OLD_PURPOSES});
const {db,users,tx,accounts,centres,period,journal,post}=m;

// Gap 1: the supplier invoice is not a ledger source; only createJournalFromPayable reaches the ledger, two lines, no VAT split.
const {payable,invoice}=m.matchedPayable({gross:'1150.00'});
m.inputTax(invoice.id,'150.00');
out('G1 SOURCE_KINDS has supplier_invoice',ledger.SOURCE_KINDS.some(k=>k.key==='supplier_invoice'));
out('G1 journalFromSource(supplier_invoice)',tryCode(()=>journal('supplier_invoice',payable.id)));
const legacy=tx(()=>f.createJournalFromPayable(db,users.employee,{payable_id:payable.id,period_id:period.id,entry_date:m.day,description:'قيد المستحق المصدر',evidence:'ربط الاختبار',debit_account_id:accounts.supplier_cost.id,credit_account_id:accounts.payable.id,cost_center_id:centres['CC-A'].id}));
out('G1 createJournalFromPayable lines [account code, debit, credit]',legacy.lines.map(l=>[db.prepare('SELECT code FROM finance_accounts WHERE id=?').get(l.account_id).code,l.debit_minor,l.credit_minor]));
const src=readFileSync(W+'/app/ledger.mjs','utf8');
const builders=src.slice(src.indexOf('function sourceDocument'),src.indexOf('export function journalFromSource'));
out("G1 source builders (sourceDocument) that post a line to 'input_vat'",(builders.match(/'input_vat'/g)??[]).length);
out('G1 sourceInvariant forces two lines',/lines\.length!==2/.test(readFileSync(W+'/app/finance.mjs','utf8')));

// Gap 2: pendingSources omits payables but lists supplier_payment; the payable GL account can go debit.
const second=m.matchedPayable({gross:'2300.00'});
m.inputTax(second.invoice.id,'300.00');
const order=m.pay(second.payable.id);
let s=ledger.statements(db,users.manager);
out('G2 pending sources kinds',[...new Set(s.sources.map(x=>x.source_kind))]);
out('G2 unposted count (two matched payables, one executed payment)',s.unposted);
post(journal('supplier_payment',order.id));
s=ledger.statements(db,users.manager);
const tb=f.listFinance(db,users.manager).trial_balance.rows.find(r=>r.code==='2000');
out('G2 payable 2000 after posting the payment only [debit balance, credit balance]',[tb.balance_debit_minor,tb.balance_credit_minor]);
out('G2 payable in balance sheet liabilities (negative = debit)',s.balance_sheet.liabilities.find(r=>r.code==='2000')?.amount_minor);
out('G2 unposted after posting the payment',s.unposted);

// Gap 3: customer advances have no builder.
const adv=m.paidAdvance({amount:'5000.00'});
for(const kind of ['advance_receipt','advance_draw','advance_reversal','customer_advance'])out(`G3 journalFromSource(${kind})`,tryCode(()=>journal(kind,adv.id)));
out("G3 'customer_advances' purpose exists",ledger.PURPOSES.some(p=>p.key==='customer_advances'));
out('G3 accruals / bank lines kinds exist',ledger.SOURCE_KINDS.filter(k=>/accru|amort|bank/.test(k.key)).map(k=>k.key));

// Gap 4/5: no control tie-out and no trace API.
out('G4 ledger exports controlReconciliation',typeof ledger.controlReconciliation);
out('G5 ledger exports traceAmount',typeof ledger.traceAmount);
out('G5 bank_matches.source_kind CHECK',db.prepare("SELECT sql FROM sqlite_master WHERE name='bank_matches'").get().sql.match(/source_kind IN \([^)]*\)/)[0]);
out('G5 legacy payable journal identity [source_kind, source_id is payable]',(()=>{const j=db.prepare('SELECT source_kind,source_id FROM finance_journals WHERE id=?').get(legacy.id);return [j.source_kind,j.source_id===payable.id];})());
out('G5 payable journal found through finance_source_links',!!db.prepare('SELECT 1 FROM finance_source_links WHERE journal_id=?').get(legacy.id));
const manual=tx(()=>f.createJournal(db,users.employee,{period_id:period.id,entry_date:m.day,description:'تسوية يدوية مصطنعة',evidence:'مذكرة مصطنعة',currency:'SAR',source_reference:'INV-2 تصحيح',lines:[{account_id:accounts.supplier_cost.id,cost_center_id:centres.GEN.id,debit:'1.00',credit:'0',memo:'تصحيح'},{account_id:accounts.payable.id,cost_center_id:centres.GEN.id,debit:'0',credit:'1.00',memo:'تصحيح'}]}));
out('G5 manual journal source [kind, id, reference]',(()=>{const j=db.prepare('SELECT source_kind,source_id,source_reference FROM finance_journals WHERE id=?').get(manual.id);return [j.source_kind,j.source_id,j.source_reference];})());

// Gap 6: constant payment_status.
out('G6 listFinance.payment_status with an executed payment order',f.listFinance(db,users.manager).payment_status);
out('G6 executed payment orders',db.prepare("SELECT COUNT(*) n FROM payment_orders WHERE status='executed'").get().n);
for(const fn of cleanups)fn();
